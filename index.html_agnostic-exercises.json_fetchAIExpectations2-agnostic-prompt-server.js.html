let editor; // Configuración de CodeMirror para ser agnóstico al lenguaje editor = CodeMirror.fromTextArea(document.getElementById('code-editor'), { lineNumbers: true, mode: 'text/x-java', // Se configura como Java para sintaxis lineWrapping: true, theme: 'seti', scrollbarStyle: 'null' }); let exercises = []; let currentExerciseIndex = 0; async function loadExercise(exercise) { try { if (!exercise || typeof exercise !== "object") { throw new Error("Objeto exercise no válido o undefined"); } const requiredProps = ['title', 'description', 'codeKoToRefactor', 'codeOkForExpect']; const missingProps = requiredProps.filter(prop => !(prop in exercise)); if (missingProps.length > 0) { throw new Error(`Ejercicio corrupto. Faltan propiedades: ${missingProps.join(', ')}`); } document.getElementById("h5_title").innerText = exercise.title; document.getElementById("challenge").innerText = exercise.description; editor.setValue(exercise.codeKoToRefactor); window.modal_click_message = exercise.modalClickMessage || "Mensaje predeterminado"; window.error_message = exercise.errorMessage || "Error predeterminado"; window.success_message = exercise.successMessage || "¡Éxito!"; if (typeof exercise.codeOkForExpect !== "string" || exercise.codeOkForExpect.trim() === "") { throw new Error("codeOkForExpect no es un código válido"); } console.log("Código para generar expectativas:", exercise.codeOkForExpect); await fetchAIResponseGetAiExpect(exercise.codeOkForExpect); } catch (error) { console.error("Error crítico en loadExercise:", error); localStorage.removeItem('testExpectations'); throw error; } } async function fetchAIResponseGetAiExpect(userCode) { try { if (typeof userCode !== "string" || userCode.trim() === "") { throw new Error(`Código inválido: ${typeof userCode}`); } const challenge = document.getElementById('challenge').textContent; const encodedCode = encodeURIComponent(userCode); const encodedChallenge = encodeURIComponent(challenge); console.log("Enviando al backend:", { code: userCode, challenge: challenge }); const response = await fetch('https://www.refactorii.com/fetchAIExpectations', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ code: encodedCode, challenge: encodedChallenge, language: "java", timestamp: new Date().toISOString() }) }); if (!response.ok) { throw new Error(`Error HTTP! estado: ${response.status}`); } const data = await response.json(); console.log("Respuesta del backend:", data); if (!data.success || !data.expectations) { throw new Error(data.message || "Respuesta del servidor inválida"); } localStorage.setItem('testExpectations', data.expectations); console.log("Expectativas almacenadas:", data.expectations); } catch (error) { console.error("Error en fetchAIResponseGetAiExpect:", error); localStorage.removeItem('testExpectations'); throw error; } } async function loadExercises(exerciseId) { try { const response = await fetch('exercises-java.json'); if (!response.ok) throw new Error(`Error al cargar JSON: ${response.status}`); const exercises = await response.json(); const selectedExercise = exercises.find(ex => String(ex.id) === String(exerciseId)); if (!selectedExercise) { throw new Error(`Ejercicio con ID ${exerciseId} no encontrado`); } if (!selectedExercise.codeOkForExpect) { throw new Error("El ejercicio no tiene codeOkForExpect"); } console.log("Ejercicio cargado:", selectedExercise); await loadExercise(selectedExercise); } catch (error) { console.error("Error crítico en loadExercises:", error); localStorage.removeItem('testExpectations'); throw error; } }
